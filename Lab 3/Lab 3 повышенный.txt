class Library:
    def _init_(self, name: str):
        self.name = name
        self.books = []  # Список для хранения названий книг

    def add_book(self, book_title: str):
        self.books.append(book_title)
        print(f'Книга "{book_title}" добавлена в библиотеку {self.name}.')

    def remove_book(self, book_title: str):
        if book_title in self.books:
            self.books.remove(book_title)
            print(f'Книга "{book_title}" удалена из библиотеки {self.name}.')
        else:
            print(f'Ошибка: Книга "{book_title}" не найдена.')

    def find_book(self, search_title: str) -> bool:
        found = search_title in self.books
        if found:
            print(f'Результат поиска: Книга "{search_title}" есть в наличии.')
        else:
            print(f'Результат поиска: Книга "{search_title}" отсутствует.')
        return found

    def list_books(self):
        print(f"Книги в библиотеке '{self.name}': {', '.join(self.books) if self.books else 'Пусто'}\n")
