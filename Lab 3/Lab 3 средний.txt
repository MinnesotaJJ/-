class BankAccount:
    def _init_(self, owner: str, initial_balance: float = 0.0):
        self.owner = owner
        self._balance = initial_balance  # Инкапсулированный атрибут

    def deposit(self, amount: float):
        if amount > 0:
            self._balance += amount
            print(f"[{self.owner}] Внесено {amount} тг. Текущий баланс: {self._balance} тг.")
        else:
            print("Сумма пополнения должна быть больше нуля.")

    def withdraw(self, amount: float):
        if amount <= 0:
            print("Сумма снятия должна быть больше нуля.")
        elif amount > self._balance:
            print(f"[{self.owner}] Ошибка: недостаточно средств!")
        else:
            self._balance -= amount
            print(f"[{self.owner}] Снято {amount} тг. Остаток: {self._balance} тг.")

    def get_balance(self) -> float:
        return self._balance
